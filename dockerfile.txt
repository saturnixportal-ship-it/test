FROM php:8.3-apache

# Install dependencies
RUN apt-get update && apt-get install -y \
    curl \
    && docker-php-ext-install mysqli pdo pdo_mysql \
    && apt-get clean

COPY . /var/www/html/
RUN chown -R www-data:www-data /var/www/html/logs && chmod 777 /var/www/html/logs

EXPOSE 80
CMD ["apache2-foreground"]
